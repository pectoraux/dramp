   252→function QueueRow({ item }: { item: OpsQueueItem }) {
   253→  const longWait = item.elapsedSeconds > 120;
   254→  const pct = Math.max(0, Math.min(100, (item.elapsedSeconds / Math.max(1, item.elapsedSeconds + item.remainingWaitSeconds)) * 100));
   255→  return (
   256→    <tr className={cn("border-b last:border-0 hover:bg-muted/30", longWait && "bg-rose-500/5")}>
   257→      <td className="px-2 py-1.5 font-mono text-[10px]">{shortId(item.executionId, 10)}</td>
   258→      <td className="px-2 py-1.5 font-mono text-[10px]">
   259→        {item.sourceAsset}:{item.sourceCountry} <ArrowRight className="inline size-2.5 mx-0.5 text-muted-foreground" /> {item.destinationAsset}:{item.destinationCountry}
   260→      </td>
   261→      <td className="px-2 py-1.5 text-right font-mono tabular-nums">{formatMoney(item.amount)} {item.sourceAsset}</td>
   262→      <td className="px-2 py-1.5"><Badge variant="outline" className="text-[10px] py-0 h-4">{prettyEnum(item.riskTolerance)}</Badge></td>
   263→      <td className="px-2 py-1.5"><Badge variant="outline" className="text-[10px] py-0 h-4">{prettyEnum(item.executionPolicy)}</Badge></td>
   264→      <td className={cn("px-2 py-1.5 text-right font-mono tabular-nums", longWait && "text-rose-600 dark:text-rose-400 font-medium")}>
   265→        {formatDuration(item.elapsedSeconds)}
   266→      </td>
   267→      <td className="px-2 py-1.5 text-right font-mono tabular-nums">{formatDuration(item.remainingWaitSeconds)}</td>
   268→      <td className="px-2 py-1.5 w-24">
   269→        <div className="h-1.5 rounded-full bg-muted overflow-hidden">
   270→          <div className={cn("h-full rounded-full", longWait ? "bg-rose-500" : "bg-amber-500")} style={{ width: `${pct}%` }} />
   271→        </div>
   272→      </td>
   273→    </tr>
   274→  );
   275→}

   654→function DisputeRow({
   655→  d,
   656→  resolving,
   657→  onResolving,

   659→}: {

   661→  resolving: boolean;
