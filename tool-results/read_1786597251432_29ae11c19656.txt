   279→function QueueRow({ item }: { item: OpsQueueItem }) {
   280→  const longWait = item.elapsedSeconds > 120;
   281→  const pct = Math.max(0, Math.min(100, (item.elapsedSeconds / Math.max(1, item.elapsedSeconds + item.remainingWaitSeconds)) * 100));
   282→  return (
   283→    <tr className={cn("border-b last:border-0 hover:bg-muted/30", longWait && "bg-rose-500/5")}>
   284→      <td className="px-2 py-1.5 font-mono text-[10px]">{shortId(item.executionId, 10)}</td>
   285→      <td className="px-2 py-1.5 font-mono text-[10px]">
   286→        {item.sourceAsset}:{item.sourceCountry} <ArrowRight className="inline size-2.5 mx-0.5 text-muted-foreground" /> {item.destinationAsset}:{item.destinationCountry}
   287→      </td>
   288→      <td className="px-2 py-1.5 text-right font-mono tabular-nums">{formatMoney(item.amount)} {item.sourceAsset}</td>
   289→      <td className="px-2 py-1.5"><Badge variant="outline" className="text-[10px] py-0 h-4">{prettyEnum(item.riskTolerance)}</Badge></td>
   290→      <td className="px-2 py-1.5"><Badge variant="outline" className="text-[10px] py-0 h-4">{prettyEnum(item.executionPolicy)}</Badge></td>
   291→      <td className={cn("px-2 py-1.5 text-right font-mono tabular-nums", longWait && "text-rose-600 dark:text-rose-400 font-medium")}>
   292→        {formatDuration(item.elapsedSeconds)}
   293→      </td>
   294→      <td className="px-2 py-1.5 text-right font-mono tabular-nums">{formatDuration(item.remainingWaitSeconds)}</td>
   295→      <td className="px-2 py-1.5 w-24">
   296→        <div className="h-1.5 rounded-full bg-muted overflow-hidden">
   297→          <div className={cn("h-full rounded-full", longWait ? "bg-rose-500" : "bg-amber-500")} style={{ width: `${pct}%` }} />
   298→        </div>
   299→      </td>
   300→    </tr>
   301→  );
   302→}

   681→function DisputeRow({
   682→  d,
   683→  resolving,
   684→  onResolving,
   685→  onResolved,
   686→}: {
   687→  d: OpsDispute;
   688→  resolving: boolean;
   689→  onResolving: (id: string | null) => void;
   690→  onResolved: () => void;
   691→}) {
   692→  const [resolution, setResolution] = useState<string>("RESOLVED_PROVIDER");

   694→  const [slashedAmount, setSlashedAmount] = useState("");

   697→
   698→  const isResolved = d.status === "RESOLVED" || !!d.resolution;
   699→
   700→  async function submit() {
   701→    onResolving(d.id);
   702→    try {
   703→      const body: any = { resolution };
   704→      if (compensationAmount) body.compensationAmount = Number(compensationAmount);
   705→      if (slashedAmount) body.slashedAmount = Number(slashedAmount);

   708→        method: "POST",
   709→        headers: { "Content-Type": "application/json" },
   710→        body: JSON.stringify(body),

   712→      const json = await res.json();

   714→      toast.success("Dispute resolved", { description: `${d.providerName} · ${resolution}` });
   715→      setOpen(false);
   716→      onResolved();

   718→      toast.error("Resolve failed", { description: err instanceof Error ? err.message : "unknown" });
   719→    } finally {
   720→      onResolving(null);
   721→    }

   723→
