     1→// Execution service — the canonical state machine + commitment model.
     2→//
     3→// State machine (happy path):
     4→//   INTENT_CREATED → SEARCHING → ROUTE_FOUND → ROUTE_RESERVED →
     5→//   ORIGIN_PENDING → ORIGIN_CONFIRMED → TOKENIZED →
     6→//   SETTLEMENT_PENDING → SETTLED →
     7→//   DESTINATION_PENDING → DESTINATION_CONFIRMED → COMPLETED
     8→//
     9→// Waiting is a POLICY, not a state: a WAIT_FOR_BETTER execution stays in
    10→// SEARCHING while dRamp continuously re-evaluates liquidity.
    11→//
    12→// Cancellation is governed by commitment_status:
    13→//   REVERSIBLE → cancel immediately, release reservations + collateral
    14→//   PARTIALLY_COMMITTED → cancellation may require resolution
    15→//   IRREVERSIBLE → cannot cancel
    16→//
    17→// Route reservation (capacity + collateral + obligations) happens in ONE
    18→// database transaction with conditional re-checks, so two simultaneous
    19→// executions cannot over-allocate one provider.
    20→

    89→export async function createIntent(input: CreateIntentInput) {
    90→  // Ensure a user exists (mocked identity).
    91→  let userId = input.userId;
    92→  if (!userId) {
    93→    const email = input.userEmail ?? "alice@dramp.demo";
    94→    const existing = await db.user.findUnique({ where: { email } });
    95→    if (existing) userId = existing.id;
    96→    else {
    97→      const u = await db.user.create({ data: { email, name: input.userName ?? "Alice" } });
    98→      userId = u.id;
    99→    }
   100→  }
   101→
   102→  const expiresAt = new Date(Date.now() + Math.max(input.maxWaitSeconds, 60) * 1000 + 60_000);
   103→  const intent = await db.executionIntent.create({
   104→    data: {
   105→      userId,
   106→      sourceAmount: new Decimal(input.sourceAmount),
   107→      sourceAsset: input.sourceAsset,
   108→      sourceCountry: input.sourceCountry,
   109→      destinationAsset: input.destinationAsset,
   110→      destinationCountry: input.destinationCountry,
   111→      minimumDestinationAmount: input.minimumDestinationAmount ? new Decimal(input.minimumDestinationAmount) : null,
   112→      maximumTotalCost: input.maximumTotalCost ? new Decimal(input.maximumTotalCost) : null,
   113→      targetRate: input.targetRate ? new Decimal(input.targetRate) : null,
   114→      riskTolerance: input.riskTolerance,
   115→      executionPolicy: input.executionPolicy,
   116→      maxWaitSeconds: input.maxWaitSeconds,
   117→      cancellationPolicy: input.cancellationPolicy ?? CANCELLATION_POLICY.CANCEL_ANYTIME_WHILE_REVERSIBLE,
   118→      allowedSettlementAssets: JSON.stringify(input.allowedSettlementAssets ?? []),
   119→      prohibitedSettlementAssets: JSON.stringify(input.prohibitedSettlementAssets ?? []),
   120→      status: "ACTIVE",
   121→      expiresAt,
   122→    },
   123→  });
   124→
   125→  // Create the first execution attempt.
   126→  const execution = await db.execution.create({
   127→    data: {
   128→      intentId: intent.id,
   129→      attemptNumber: 1,
   130→      status: EXECUTION_STATUS.INTENT_CREATED,
   131→      commitmentStatus: COMMITMENT_STATUS.REVERSIBLE,
   132→      startedAt: new Date(),
   133→      lastTickAt: new Date(),
   134→    },
   135→  });
   136→
   137→  await appendAuditEvent({
   138→    intentId: intent.id,
   139→    executionId: execution.id,
   140→    eventType: "intent_created",
   141→    payload: {
   142→      sourceAmount: input.sourceAmount,
   143→      sourceAsset: input.sourceAsset,
   144→      destinationAsset: input.destinationAsset,
   145→      riskTolerance: input.riskTolerance,
   146→      executionPolicy: input.executionPolicy,
   147→      maxWaitSeconds: input.maxWaitSeconds,
   148→    },
   149→    actorType: AUDIT_ACTOR.USER,
   150→    actorId: userId,
   151→  });
   152→
   153→  // Immediately move to SEARCHING and discover routes.
   154→  await db.execution.update({
   155→    where: { id: execution.id },
   156→    data: { status: EXECUTION_STATUS.SEARCHING, lastTickAt: new Date() },
   157→  });
   158→  await appendAuditEvent({
   159→    executionId: execution.id,
   160→    eventType: "execution_searching",
   161→    payload: { policy: input.executionPolicy },
   162→  });
   163→
   164→  const routes = await discoverAndPersistRoutes(execution.id, intent);
   165→
   166→  // If NOW policy and a valid route exists, select it immediately.
   167→  if (input.executionPolicy === EXECUTION_POLICY.NOW) {
   168→    const best = routes.find((r) => r.tag === ROUTE_TAG.BEST);
   169→    if (best) {
   170→      await selectRoute(execution.id, best.id);
   171→    }
   172→  } else if (input.executionPolicy === EXECUTION_POLICY.WAIT_FOR_BETTER) {
   173→    // Set the best valid route as the reference; stay in SEARCHING.
   174→    const best = routes.find((r) => !r.hardFilterRejection);
   175→    if (best) {
   176→      await db.execution.update({
   177→        where: { id: execution.id },
   178→        data: { referenceRouteId: best.id },
   179→      });
   180→      await appendAuditEvent({
   181→        executionId: execution.id,
   182→        eventType: "reference_route_set",
   183→        payload: { routeId: best.id, tag: best.tag, effectiveCost: best.effectiveCost.toString() },
   184→      });
   185→    }
   186→  }
   187→
   188→  return { intent, execution, routes };
   189→}

   193→export async function discoverAndPersistRoutes(executionId: string, intent: { id: string; sourceAmount: Prisma.Decimal; sourceAsset: string; sourceCountry: string; destinationAsset: string; destinationCountry: string; riskTolerance: string; allowedSettlementAssets: string; prohibitedSettlementAssets: string }) {
   194→  const allowed = safeParse(intent.allowedSettlementAssets);
   195→  const prohibited = safeParse(intent.prohibitedSettlementAssets);
   196→  const candidates = await findRoutes({
   197→    sourceAmount: intent.sourceAmount,
   198→    sourceAsset: intent.sourceAsset,
   199→    sourceCountry: intent.sourceCountry,
   200→    destinationAsset: intent.destinationAsset,
   201→    destinationCountry: intent.destinationCountry,
   202→    riskTolerance: intent.riskTolerance,
   203→    allowedSettlementAssets: allowed,
   204→    prohibitedSettlementAssets: prohibited,
   205→  });
   206→
   207→  const persisted: Array<CandidateRoute & { id: string }> = [];
   208→  for (const c of candidates) {
   209→    const route = await db.route.create({
   210→      data: {
   211→        executionId,
   212→        legCount: c.hopCount,
   213→        totalCost: c.totalCost,
   214→        effectiveCost: c.effectiveCost,
   215→        grossOutput: c.grossOutput,
   216→        netOutput: c.netOutput,
   217→        incentiveBps: c.incentiveBps,
   218→        riskCounterparty: c.risk.counterparty,
   219→        riskSettlementAsset: c.risk.settlementAsset,
   220→        riskLiquidity: c.risk.liquidity,
   221→        riskOperational: c.risk.operational,
   222→        riskDuration: c.risk.duration,
   223→        riskComposite: c.risk.composite,
   224→        expectedExecutionSeconds: c.expectedExecutionSeconds,
   225→        explanation: c.explanation || (c.hardFilterRejection ?? "Candidate route"),
   226→        tag: c.tag,
   227→        status: c.hardFilterRejection ? ROUTE_STATUS.FAILED : ROUTE_STATUS.CANDIDATE,
   228→        splitRoute: c.split,
   229→        expiresAt: new Date(Date.now() + 60_000),
   230→      },
   231→    });
   232→    for (const leg of c.legs) {
   233→      await db.leg.create({
   234→        data: {
   235→          routeId: route.id,
   236→          executionId,
   237→          providerId: leg.providerId,
   238→          offerId: leg.offerId,
   239→          sequence: leg.sequence,
   240→          role: leg.role,
   241→          amount: leg.amount,
   242→          sourceAsset: leg.sourceAsset,
   243→          destinationAsset: leg.destinationAsset,
   244→          settlementAssetId: leg.settlementAssetId,
   245→          channelType: leg.channelType,
   246→          status: LEG_STATUS.PENDING,
   247→          commitmentStatus: COMMITMENT_STATUS.REVERSIBLE,
   248→        },
   249→      });
   250→    }
   251→    persisted.push({ ...c, id: route.id });
   252→  }
   253→
   254→  await appendAuditEvent({
   255→    executionId,
   256→    eventType: "routes_discovered",
   257→    payload: {
   258→      count: candidates.length,
   259→      valid: candidates.filter((c) => !c.hardFilterRejection).length,
   260→      rejected: candidates.filter((c) => c.hardFilterRejection).map((c) => ({ reason: c.hardFilterRejection })),
   261→    },
   262→  });
   263→
   264→  return persisted;
   265→}

   269→export async function selectRoute(executionId: string, routeId: string) {
   270→  const execution = await db.execution.findUnique({ where: { id: executionId } });
   271→  if (!execution) throw new Error("Execution not found");
   272→  if (execution.status !== EXECUTION_STATUS.SEARCHING && execution.status !== EXECUTION_STATUS.ROUTE_FOUND) {
   273→    throw new Error(`Cannot select route in state ${execution.status}`);
   274→  }
   275→  await db.route.updateMany({
   276→    where: { executionId, status: ROUTE_STATUS.CANDIDATE },
   277→    data: { status: ROUTE_STATUS.SUPERSEDED },
   278→  });
   279→  await db.route.update({
   280→    where: { id: routeId },
   281→    data: { status: ROUTE_STATUS.SELECTED },
   282→  });
   283→  const snapshot = await buildRouteSnapshot(routeId);
   284→  await db.execution.update({
   285→    where: { id: executionId },
   286→    data: {
   287→      status: EXECUTION_STATUS.ROUTE_FOUND,
   288→      selectedRouteId: routeId,
   289→      selectedRouteJson: JSON.stringify(snapshot),
   290→      lastTickAt: new Date(),
   291→    },
   292→  });
   293→  await appendAuditEvent({
   294→    executionId,
   295→    eventType: "route_selected",
   296→    payload: { routeId, tag: snapshot.tag, netOutput: snapshot.netOutput, effectiveCost: snapshot.effectiveCost },
   297→  });
   298→}

   300→// ---- Route reservation (the critical concurrency point) ------------------
   301→
   302→export async function reserveRoute(executionId: string) {
   303→  const execution = await db.execution.findUnique({
   304→    where: { id: executionId },
   305→    include: { intent: true },
   306→  });
   307→  if (!execution) throw new Error("Execution not found");
   308→  if (execution.status !== EXECUTION_STATUS.ROUTE_FOUND) {
   309→    throw new Error(`Cannot reserve in state ${execution.status}`);
   310→  }
   311→  const routeId = execution.selectedRouteId;
   312→  if (!routeId) throw new Error("No route selected");
   313→
   314→  // Run the whole reservation in a single interactive transaction.
   315→  await db.$transaction(async (tx) => {
   316→    const route = await tx.route.findUnique({
   317→      where: { id: routeId },
   318→      include: { legs: { include: { provider: true, offer: true } } },
   319→    });
   320→    if (!route) throw new Error("Route not found");
   321→
   322→    // 1. Reserve capacity on each offer (conditional updates).
   323→    // 2. For collateralized providers, lock collateral (assert eligibility).
   324→    // 3. Create obligations.
   325→    // 4. Update execution + route status.
   326→    const expiresAt = new Date(Date.now() + 5 * 60_000);
   327→    const obligationIds: string[] = [];
   328→
   329→    // Group legs by sequence to handle split routes.
   330→    const bySequence = new Map<number, typeof route.legs>();
   331→    for (const leg of route.legs) {
   332→      if (!bySequence.has(leg.sequence)) bySequence.set(leg.sequence, []);
   333→      bySequence.get(leg.sequence)!.push(leg);
   334→    }
   335→    const sequences = [...bySequence.keys()].sort((a, b) => a - b);
   336→
   337→    for (const seq of sequences) {
   338→      const legs = bySequence.get(seq)!;
   339→      for (const leg of legs) {
   340→        // Reserve capacity.
   341→        await reserveCapacity({
   342→          offerId: leg.offerId,
   343→          executionId,
   344→          amount: leg.amount,
   345→          expiresAt,
   346→          idempotencyKey: `reserve:${executionId}:${leg.id}`,
   347→          tx,
   348→        });
   349→
   350→        // Update leg status.
   351→        await tx.leg.update({
   352→          where: { id: leg.id },
   353→          data: { status: LEG_STATUS.RESERVED },
   354→        });
   355→
   356→        // Lock collateral if the provider is collateralized.
   357→        if (leg.provider.trustModel === "COLLATERALIZED") {
   358→          const vault = await tx.vault.findUnique({ where: { providerId: leg.providerId } });
   359→          if (!vault) throw new Error(`Collateralized provider ${leg.provider.name} has no vault`);
   360→          // Determine the collateral asset (first stable holding).
   361→          const holdings = JSON.parse(vault.holdingsJson) as { asset: string; amount: string }[];
   362→          const collateralAsset = holdings[0]?.asset;
   363→          if (!collateralAsset) throw new Error(`Vault for ${leg.provider.name} has no holdings`);
   364→          const sa = await tx.settlementAsset.findUnique({ where: { symbol: collateralAsset } });
   365→          if (!sa) throw new Error(`Collateral asset ${collateralAsset} not found`);
   366→          assertCollateralEligible(sa);
   367→          const { lock: lk } = await lockCollateral({
   368→            providerId: leg.providerId,
   369→            executionId,
   370→            amount: leg.amount,
   371→            asset: collateralAsset,
   372→            idempotencyKey: `lock:${executionId}:${leg.id}`,
   373→            tx,
   374→          });
   375→
   376→          // Post a LOCK ledger entry.
   377→          await tx.ledgerEntry.create({
   378→            data: {
   379→              debitAccount: `provider:${leg.providerId}:vault:${collateralAsset}`,
   380→              creditAccount: `provider:${leg.providerId}:operational:${collateralAsset}`,
   381→              amount: leg.amount,
   382→              asset: collateralAsset,
   383→              entryType: "LOCK",
   384→              executionId,
   385→              idempotencyKey: `ledger-lock:${executionId}:${leg.id}`,
   386→              description: `Collateral locked for leg ${leg.id}`,
   387→            },
   388→          });
   389→          void lk;
   390→        }
   391→
   392→        // Create the obligation.
   393→        const obligation = await tx.obligation.create({
   394→          data: {
   395→            executionId,
   396→            legId: leg.id,
   397→            providerId: leg.providerId,
   398→            amount: leg.amount,
   399→            asset: leg.sourceAsset,
   400→            dueAt: new Date(Date.now() + 10 * 60_000),
   401→            status: OBLIGATION_STATUS.CREATED,
   402→            collateralBackingId: null,
   403→          },
   404→        });
   405→        obligationIds.push(obligation.id);
   406→      }
   407→    }
   408→
   409→    // Update execution + route.
   410→    await tx.execution.update({
   411→      where: { id: executionId },
   412→      data: {
   413→        status: EXECUTION_STATUS.ROUTE_RESERVED,
   414→        commitmentStatus: COMMITMENT_STATUS.REVERSIBLE,
   415→        lastTickAt: new Date(),
   416→      },
   417→    });
   418→    await tx.route.update({
   419→      where: { id: routeId },
   420→      data: { status: ROUTE_STATUS.RESERVED },
   421→    });
   422→  }, { timeout: 30000, maxWait: 15000 });
   423→
   424→  await appendAuditEvent({
   425→    executionId,
   426→    eventType: "route_reserved",
   427→    payload: { routeId },
   428→  });
   429→  await appendAuditEvent({
   430→    executionId,
   431→    eventType: "obligations_created",
   432→    payload: { count: obligationIdsPlaceholder(executionId) },
   433→  });
   434→}

   442→// ---- State advancement (driven by the ticker) ----------------------------
   443→
   444→export async function advanceExecution(executionId: string): Promise<{ advanced: boolean; reason?: string }> {
   445→  const execution = await db.execution.findUnique({
   446→    where: { id: executionId },
   447→    include: { intent: true },
   448→  });
   449→  if (!execution) return { advanced: false, reason: "not found" };
   450→  if (TERMINAL_STATES.has(execution.status)) return { advanced: false, reason: "terminal" };
   451→
   452→  // Check intent expiry.
   453→  if (execution.intent.status === "EXPIRED" || execution.intent.expiresAt < new Date()) {
   454→    if (execution.status === EXECUTION_STATUS.SEARCHING || execution.status === EXECUTION_STATUS.INTENT_CREATED) {
   455→      await expireExecution(executionId, "intent expired");
   456→      return { advanced: true, reason: "expired" };
   457→    }
   458→  }
   459→
   460→  // Enforce dwell time so the UI can render each state.
   461→  const dwellElapsed = Date.now() - execution.lastTickAt.getTime() >= STATE_DWELL_MS;
   462→  if (!dwellElapsed) return { advanced: false, reason: "dwell" };
   463→
   464→  switch (execution.status) {
   465→    case EXECUTION_STATUS.SEARCHING:
   466→      return advanceSearching(execution);
   467→    case EXECUTION_STATUS.ROUTE_FOUND:
   468→      await reserveRoute(executionId);
   469→      return { advanced: true, reason: "reserved" };
   470→    case EXECUTION_STATUS.ROUTE_RESERVED:
   471→      await beginOriginPending(executionId);
   472→      return { advanced: true, reason: "origin_pending" };
   473→    case EXECUTION_STATUS.ORIGIN_PENDING:
   474→      // Simulate the user's fiat-in payment arriving.
   475→      await confirmOrigin(executionId);
   476→      return { advanced: true, reason: "origin_confirmed" };
   477→    case EXECUTION_STATUS.ORIGIN_CONFIRMED:
   478→      await tokenize(executionId);
   479→      return { advanced: true, reason: "tokenized" };
   480→    case EXECUTION_STATUS.TOKENIZED:
   481→      await beginSettlement(executionId);
   482→      return { advanced: true, reason: "settlement_pending" };
   483→    case EXECUTION_STATUS.SETTLEMENT_PENDING:
   484→      await settle(executionId);
   485→      return { advanced: true, reason: "settled" };
   486→    case EXECUTION_STATUS.SETTLED:
   487→      await beginDestination(executionId);
   488→      return { advanced: true, reason: "destination_pending" };
   489→    case EXECUTION_STATUS.DESTINATION_PENDING:
   490→      return await confirmDestination(executionId);
   491→    case EXECUTION_STATUS.DESTINATION_CONFIRMED:
   492→      await completeExecution(executionId);
   493→      return { advanced: true, reason: "completed" };
   494→    default:
   495→      return { advanced: false, reason: `unhandled ${execution.status}` };
   496→  }
   497→}

   499→// ---- SEARCHING (waiting / re-evaluation) ---------------------------------
   500→
   501→async function advanceSearching(execution: Prisma.ExecutionGetPayload<{ include: { intent: true } }>) {
   502→  // Re-evaluate routes in-memory (do NOT re-persist every tick — that would
   503→  // duplicate route records and break the reference lookup).
   504→  const candidates = await findRoutes({
   505→    sourceAmount: execution.intent.sourceAmount,
   506→    sourceAsset: execution.intent.sourceAsset,
   507→    sourceCountry: execution.intent.sourceCountry,
   508→    destinationAsset: execution.intent.destinationAsset,
   509→    destinationCountry: execution.intent.destinationCountry,
   510→    riskTolerance: execution.intent.riskTolerance,
   511→    allowedSettlementAssets: safeParse(execution.intent.allowedSettlementAssets),
   512→    prohibitedSettlementAssets: safeParse(execution.intent.prohibitedSettlementAssets),
   513→  });
   514→
   515→  const validCandidates = candidates.filter((c) => !c.hardFilterRejection);
   516→  if (validCandidates.length === 0) {
   517→    if (Date.now() - execution.startedAt.getTime() > execution.intent.maxWaitSeconds * 1000) {
   518→      await expireExecution(execution.id, "no valid route found within max wait");
   519→      return { advanced: true, reason: "expired_no_route" };
   520→    }
   521→    await db.execution.update({ where: { id: execution.id }, data: { lastTickAt: new Date() } });
   522→    return { advanced: false, reason: "no_valid_routes" };
   523→  }
   524→
   525→  const best = validCandidates[0]; // already sorted by rankAndTag
   526→
   527→  if (execution.intent.executionPolicy === EXECUTION_POLICY.NOW) {
   528→    // Persist the selected route and move to ROUTE_FOUND.
   529→    const persisted = await persistSingleRoute(execution.id, best);
   530→    await selectRoute(execution.id, persisted.id);
   531→    return { advanced: true, reason: "now_selected" };
   532→  }
   533→
   534→  // WAIT_FOR_BETTER: compare the in-memory best to the persisted reference.
   535→  const refRoute = execution.referenceRouteId
   536→    ? await db.route.findUnique({ where: { id: execution.referenceRouteId } })
   537→    : null;
   538→
   539→  const waitedSeconds = Math.floor((Date.now() - execution.startedAt.getTime()) / 1000);
   540→  const timeout = waitedSeconds >= execution.intent.maxWaitSeconds;
   541→
   542→  if (!refRoute) {
   543→    // Persist the best as the initial reference.
   544→    const persisted = await persistSingleRoute(execution.id, best);
   545→    await db.execution.update({
   546→      where: { id: execution.id },
   547→      data: { referenceRouteId: persisted.id, lastTickAt: new Date() },
   548→    });
   549→    await appendAuditEvent({
   550→      executionId: execution.id,
   551→      eventType: "reference_route_set",
   552→      payload: { routeId: persisted.id, tag: best.tag, effectiveCost: best.effectiveCost.toString() },
   553→    });
   554→    return { advanced: false, reason: "reference_set" };
   555→  }
   556→
   557→  // Improvement = how much cheaper (in bps of notional) the new best is.
   558→  const improvementBps = refRoute.effectiveCost
   559→    .minus(best.effectiveCost)
   560→    .dividedBy(execution.intent.sourceAmount)
   561→    .times(10000)
   562→    .toNumber();
   563→
   564→  await db.execution.update({
   565→    where: { id: execution.id },
   566→    data: { waitedSeconds, lastTickAt: new Date() },
   567→  });
   568→
   569→  const improved = improvementBps >= 3; // 3 bps improvement threshold
   570→
   571→  if (improved) {
   572→    // Persist the new best, update reference, and execute.
   573→    const persisted = await persistSingleRoute(execution.id, best);
   574→    await db.execution.update({
   575→      where: { id: execution.id },
   576→      data: { referenceRouteId: persisted.id },
   577→    });
   578→    await appendAuditEvent({
   579→      executionId: execution.id,
   580→      eventType: "better_route_found",
   581→      payload: {
   582→        previousRouteId: refRoute.id,
   583→        newRouteId: persisted.id,
   584→        improvementBps: Number(improvementBps.toFixed(2)),
   585→        previousEffectiveCost: refRoute.effectiveCost.toString(),
   586→        newEffectiveCost: best.effectiveCost.toString(),
   587→      },
   588→    });
   589→    await selectRoute(execution.id, persisted.id);
   590→    return { advanced: true, reason: "better_route_selected" };
   591→  }
   592→
   593→  if (timeout) {
   594→    const persisted = await persistSingleRoute(execution.id, best);
   595→    await appendAuditEvent({
   596→      executionId: execution.id,
   597→      eventType: "wait_timeout_executing_best",
   598→      payload: { routeId: persisted.id, waitedSeconds },
   599→    });
   600→    await selectRoute(execution.id, persisted.id);
   601→    return { advanced: true, reason: "timeout_executed" };
   602→  }
   603→
   604→  return { advanced: false, reason: `waiting_improvement=${improvementBps.toFixed(2)}bps` };
   605→}

   712→async function tokenize(executionId: string) {
   713→  const execution = await db.execution.findUnique({
   714→    where: { id: executionId },
   715→    include: { intent: true },
   716→  });
   717→  if (!execution) throw new Error("Execution not found");
   718→  const route = await fetchRouteWithLegs(execution.selectedRouteId);
   719→  if (!route) throw new Error("Selected route not found");
   720→  // Source legs convert source asset -> settlement asset (or directly to dest).
   721→  const sourceLegs = route.legs
   722→    .filter((l) => l.role === LEG_ROLE.SOURCE)
   723→    .sort((a, b) => a.sequence - b.sequence);
   724→
   725→  for (const leg of sourceLegs) {
   726→    // Move source asset from escrow to provider operational.
   727→    await transfer(
   728→      `execution:${executionId}:escrow:${leg.sourceAsset}`,
   729→      `provider:${leg.providerId}:operational:${leg.sourceAsset}`,
   730→      leg.amount,
   731→      leg.sourceAsset,
   732→      { executionId, idempotencyKey: `tokenize-in:${executionId}:${leg.id}`, description: `Source funds to ${leg.provider.name}` },
   733→    );
   734→    // Charge the fee.
   735→    const feeAmt = feeForAmount(leg.amount, leg.offer.feeBps);
   736→    if (feeAmt.gt(0)) {
   737→      await feeEntry(
   738→        `provider:${leg.providerId}:operational:${leg.sourceAsset}`,
   739→        feeAmt,
   740→        leg.sourceAsset,
   741→        { executionId, idempotencyKey: `fee:${executionId}:${leg.id}`, description: `Provider fee ${leg.offer.feeBps}bps` },
   742→      );
   743→    }
   744→    // Mint / credit the settlement asset (or destination asset) to escrow.
   745→    const outAmount = moneySub(moneyMul(moneySub(leg.amount, feeAmt), leg.offer.rate), new Decimal(0));
   746→    await mint(
   747→      `execution:${executionId}:escrow:${leg.destinationAsset}`,
   748→      outAmount,
   749→      leg.destinationAsset,
   750→      { executionId, idempotencyKey: `mint:${executionId}:${leg.id}`, description: `Tokenized ${leg.destinationAsset} from ${leg.provider.name}` },
   751→    );
   752→    // Apply incentive if any (rebate to execution escrow).
   753→    if (leg.offer.incentiveBps > 0) {
   754→      const inc = incentiveForAmount(outAmount, leg.offer.incentiveBps);
   755→      await incentiveEntry(
   756→        `execution:${executionId}:escrow:${leg.destinationAsset}`,
   757→        inc,
   758→        leg.destinationAsset,
   759→        { executionId, idempotencyKey: `incentive:${executionId}:${leg.id}`, description: `Settlement incentive ${leg.offer.incentiveBps}bps` },
   760→      );
   761→    }
   762→    await db.leg.update({ where: { id: leg.id }, data: { status: LEG_STATUS.CONFIRMED, confirmedAt: new Date() } });
   763→  }
   764→
   765→  await db.execution.update({
   766→    where: { id: executionId },
   767→    data: { status: EXECUTION_STATUS.TOKENIZED, lastTickAt: new Date() },
   768→  });
   769→  await appendAuditEvent({
   770→    executionId,
   771→    eventType: "tokenized",
   772→    payload: { legs: sourceLegs.map((l) => ({ id: l.id, provider: l.provider.name, asset: l.destinationAsset })) },
   773→  });
   774→}

   808→async function settle(executionId: string) {
   809→  const execution = await db.execution.findUnique({ where: { id: executionId } });
   810→  if (!execution) throw new Error("Execution not found");
   811→  const route = await fetchRouteWithLegs(execution.selectedRouteId);
   812→  if (!route) throw new Error("Selected route not found");
   813→  const hopLegs = route.legs
   814→    .filter((l) => l.role === LEG_ROLE.SETTLEMENT_HOP)
   815→    .sort((a, b) => a.sequence - b.sequence);
   816→  for (const leg of hopLegs) {
   817→    // Burn the consumed settlement asset and mint the next asset to escrow.
   818→    await burn(
   819→      `provider:${leg.providerId}:operational:${leg.sourceAsset}`,
   820→      leg.amount,
   821→      leg.sourceAsset,
   822→      { executionId, idempotencyKey: `burn:${executionId}:${leg.id}`, description: `Settlement asset consumed by ${leg.provider.name}` },
   823→    );
   824→    const feeAmt = feeForAmount(leg.amount, leg.offer.feeBps);
   825→    if (feeAmt.gt(0)) {
   826→      await feeEntry(
   827→        `provider:${leg.providerId}:operational:${leg.sourceAsset}`,
   828→        feeAmt,
   829→        leg.sourceAsset,
   830→        { executionId, idempotencyKey: `fee:${executionId}:${leg.id}`, description: `Provider fee` },
   831→      );
   832→    }
   833→    const outAmount = moneyMul(moneySub(leg.amount, feeAmt), leg.offer.rate);
   834→    await mint(
   835→      `execution:${executionId}:escrow:${leg.destinationAsset}`,
   836→      outAmount,
   837→      leg.destinationAsset,
   838→      { executionId, idempotencyKey: `mint:${executionId}:${leg.id}`, description: `Settlement hop output` },
   839→    );
   840→    await db.leg.update({ where: { id: leg.id }, data: { status: LEG_STATUS.CONFIRMED, confirmedAt: new Date() } });
   841→  }
   842→  await db.execution.update({
   843→    where: { id: executionId },
   844→    data: { status: EXECUTION_STATUS.SETTLED, commitmentStatus: COMMITMENT_STATUS.IRREVERSIBLE, lastTickAt: new Date() },
   845→  });
   846→  await appendAuditEvent({
   847→    executionId,
   848→    eventType: "settlement_completed",
   849→    payload: {},
   850→  });
   851→}

   871→async function confirmDestination(executionId: string): Promise<{ advanced: boolean; reason?: string }> {
   872→  const execution = await db.execution.findUnique({ where: { id: executionId } });
   873→  if (!execution) throw new Error("Execution not found");
   874→  const route = await fetchRouteWithLegs(execution.selectedRouteId);
   875→  if (!route) throw new Error("Selected route not found");
   876→  const destLegs = route.legs
   877→    .filter((l) => l.role === LEG_ROLE.DESTINATION)
   878→    .sort((a, b) => a.sequence - b.sequence);
   879→
   880→  // For automatic legs, attempt confirmation. For manual legs, wait for the
   881→  // provider console to confirm.
   882→  for (const leg of destLegs) {
   883→    if (leg.status === LEG_STATUS.CONFIRMED) continue;
   884→    if (leg.channelType === CHANNEL_TYPE.AUTOMATIC) {
   885→      const r = await attemptAutomaticConfirmation(leg.id);
   886→      if (!r.confirmed) {
   887→        // Auto-confirmation failed — surface as a dispute/failure path.
   888→        await db.execution.update({ where: { id: executionId }, data: { lastTickAt: new Date() } });
   889→        await appendAuditEvent({
   890→          executionId,
   891→          eventType: "destination_auto_confirm_failed",
   892→          payload: { legId: leg.id, reason: r.reason },
   893→        });
   894→        return { advanced: false, reason: "auto_confirm_failed" };
   895→      }
   896→    } else {
   897→      // Manual: check if confirmed via console.
   898→      if (leg.status !== LEG_STATUS.CONFIRMED) {
   899→        await db.execution.update({ where: { id: executionId }, data: { lastTickAt: new Date() } });
   900→        return { advanced: false, reason: "awaiting_manual_confirmation" };
   901→      }
   902→    }
   903→  }
   904→
   905→  // All destination legs confirmed — pay out to recipient.
   906→  for (const leg of destLegs) {
   907→    // Move settlement asset from escrow to provider, then burn/mint payout.
   908→    await transfer(
   909→      `execution:${executionId}:escrow:${leg.sourceAsset}`,
   910→      `provider:${leg.providerId}:operational:${leg.sourceAsset}`,
   911→      leg.amount,
   912→      leg.sourceAsset,
   913→      { executionId, idempotencyKey: `dest-in:${executionId}:${leg.id}`, description: `Funds to payout provider ${leg.provider.name}` },
   914→    );
   915→    const feeAmt = feeForAmount(leg.amount, leg.offer.feeBps);
   916→    if (feeAmt.gt(0)) {
   917→      await feeEntry(
   918→        `provider:${leg.providerId}:operational:${leg.sourceAsset}`,
   919→        feeAmt,
   920→        leg.sourceAsset,
   921→        { executionId, idempotencyKey: `fee:${executionId}:${leg.id}`, description: `Payout provider fee` },
   922→      );
   923→    }
   924→    const payout = moneyMul(moneySub(leg.amount, feeAmt), leg.offer.rate);
   925→    await mint(
   926→      `execution:${executionId}:escrow:${leg.destinationAsset}`,
   927→      payout,
   928→      leg.destinationAsset,
   929→      { executionId, idempotencyKey: `payout:${executionId}:${leg.id}`, description: `Payout minted` },
   930→    );
   931→    await transfer(
   932→      `execution:${executionId}:escrow:${leg.destinationAsset}`,
   933→      `external:destination:${leg.destinationAsset}`,
   934→      payout,
   935→      leg.destinationAsset,
   936→      { executionId, idempotencyKey: `payout-out:${executionId}:${leg.id}`, description: `Payout to recipient` },
   937→    );
   938→  }
   939→
   940→  await db.execution.update({
   941→    where: { id: executionId },
   942→    data: { status: EXECUTION_STATUS.DESTINATION_CONFIRMED, lastTickAt: new Date() },
   943→  });
   944→  await appendAuditEvent({
   945→    executionId,
   946→    eventType: "payout_completed",
   947→    payload: {},
   948→  });
   949→  return { advanced: true, reason: "destination_confirmed" };
   950→}

   954→async function completeExecution(executionId: string) {
   955→  const execution = await db.execution.findUnique({
   956→    where: { id: executionId },
   957→    include: { intent: true },
   958→  });
   959→  if (!execution) throw new Error("Execution not found");
   960→
   961→  // Release collateral, fulfill obligations, consume reservations.
   962→  await db.$transaction(async (tx) => {
   963→    await releaseCollateralForExecution(executionId, tx);
   964→    await consumeReservationsForExecution(executionId, tx);
   965→    // Fulfill all obligations.
   966→    const obligations = await tx.obligation.findMany({ where: { executionId } });
   967→    for (const ob of obligations) {
   968→      await tx.obligation.update({
   969→        where: { id: ob.id },
   970→        data: { status: OBLIGATION_STATUS.FULFILLED, fulfilledAt: new Date() },
   971→      });
   972→    }
   973→    // Release collateral ledger entries.
   974→    const locks = await tx.collateralLock.findMany({ where: { executionId } });
   975→    for (const lk of locks) {
   976→      await tx.ledgerEntry.create({
   977→        data: {
   978→          debitAccount: `provider:${lk.providerId}:operational:${lk.asset}`,
   979→          creditAccount: `provider:${lk.providerId}:vault:${lk.asset}`,
   980→          amount: lk.amount,
   981→          asset: lk.asset,
   982→          entryType: "RELEASE",
   983→          executionId,
   984→          idempotencyKey: `release:${executionId}:${lk.id}`,
   985→          description: "Collateral released after completion",
   986→        },
   987→      });
   988→    }
   989→    // Mark remaining legs confirmed.
   990→    await tx.leg.updateMany({
   991→      where: { executionId, status: { not: LEG_STATUS.CONFIRMED } },
   992→      data: { status: LEG_STATUS.CONFIRMED, commitmentStatus: COMMITMENT_STATUS.IRREVERSIBLE },
   993→    });
   994→    await tx.execution.update({
   995→      where: { id: executionId },
   996→      data: { status: EXECUTION_STATUS.COMPLETED, commitmentStatus: COMMITMENT_STATUS.IRREVERSIBLE, completedAt: new Date(), lastTickAt: new Date() },
   997→    });
   998→    await tx.executionIntent.update({
   999→      where: { id: execution.intentId },
  1000→      data: { status: "COMPLETED" },
  1001→    });
  1002→  }, { timeout: 30000, maxWait: 15000 });
  1003→
  1004→  await appendAuditEvent({
  1005→    executionId,
  1006→    eventType: "execution_completed",
  1007→    payload: {},
  1008→  });
  1009→}

  1011→// ---- Cancellation --------------------------------------------------------

  1013→export async function cancelExecution(executionId: string, actorId?: string): Promise<{ cancelled: boolean; reason?: string }> {
  1014→  const execution = await db.execution.findUnique({ where: { id: executionId } });
  1015→  if (!execution) throw new Error("not found");
  1016→  if (TERMINAL_STATES.has(execution.status)) {
  1017→    return { cancelled: false, reason: `already terminal (${execution.status})` };
  1018→  }
  1019→  if (execution.commitmentStatus === COMMITMENT_STATUS.IRREVERSIBLE) {
  1020→    return { cancelled: false, reason: "Execution is IRREVERSIBLE — cancellation not possible" };
  1021→  }
  1022→  if (execution.commitmentStatus === COMMITMENT_STATUS.PARTIALLY_COMMITTED) {
  1023→    // Cancellation may require resolution; for the prototype we allow it but
  1024→    // initiate a refund of any received funds.
  1025→    await db.$transaction(async (tx) => {
  1026→      await releaseReservationsForExecution(executionId, tx);
  1027→      await releaseCollateralForExecution(executionId, tx);
  1028→      await tx.leg.updateMany({
  1029→        where: { executionId },
  1030→        data: { status: LEG_STATUS.CANCELLED },
  1031→      });
  1032→      await tx.execution.update({
  1033→        where: { id: executionId },
  1034→        data: { status: EXECUTION_STATUS.CANCELLED, completedAt: new Date() },
  1035→      });
  1036→    }, { timeout: 30000, maxWait: 15000 });
  1037→    await appendAuditEvent({
  1038→      executionId,
  1039→      eventType: "execution_cancelled",
  1040→      payload: { commitment: COMMITMENT_STATUS.PARTIALLY_COMMITTED, note: "Partial-commitment cancellation with refund resolution" },
  1041→      actorType: AUDIT_ACTOR.USER,
  1042→      actorId,
  1043→    });
  1044→    return { cancelled: true, reason: "partial_commitment_refund" };
  1045→  }
  1046→
  1047→  // Fully reversible.
  1048→  await db.$transaction(async (tx) => {
  1049→    await releaseReservationsForExecution(executionId, tx);
  1050→    await releaseCollateralForExecution(executionId, tx);
  1051→    await tx.leg.updateMany({
  1052→      where: { executionId },
  1053→      data: { status: LEG_STATUS.CANCELLED },
  1054→    });
  1055→    await tx.execution.update({
  1056→      where: { id: executionId },
  1057→      data: { status: EXECUTION_STATUS.CANCELLED, completedAt: new Date() },
  1058→    });
  1059→  }, { timeout: 30000, maxWait: 15000 });
  1060→  await appendAuditEvent({
  1061→    executionId,
  1062→    eventType: "execution_cancelled",
  1063→    payload: { commitment: COMMITMENT_STATUS.REVERSIBLE },
  1064→    actorType: AUDIT_ACTOR.USER,
  1065→    actorId,
  1066→  });
  1067→  return { cancelled: true, reason: "reversible_cancelled" };
  1068→}
